cmake_minimum_required(VERSION 3.24)
project(QuickClipboard VERSION 1.0.0 LANGUAGES C CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "QuickClipboard uses Win32 and DPAPI; build on Windows (x64).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
add_library(sqlite3 STATIC vendor/sqlite/sqlite3.c)
target_include_directories(sqlite3 PUBLIC vendor/sqlite)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION SQLITE_DEFAULT_MEMSTATUS=0 SQLITE_DQS=0)
add_library(quickclipboard_core STATIC src/core.cpp src/database.cpp src/hotkey.cpp src/ui.cpp src/startup.cpp)
target_include_directories(quickclipboard_core PUBLIC src)
target_link_libraries(quickclipboard_core PUBLIC sqlite3 crypt32 bcrypt shell32 advapi32 comctl32)
add_executable(QuickClipboard WIN32 src/main.cpp src/app.cpp src/quickclipboard.rc)
target_include_directories(QuickClipboard PRIVATE assets)
target_link_libraries(QuickClipboard PRIVATE quickclipboard_core comctl32 dwmapi shlwapi uxtheme)
if(MSVC)
  target_link_options(QuickClipboard PRIVATE /MANIFEST:NO)
endif()
if(MINGW)
  target_link_options(QuickClipboard PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
option(QUICKCLIPBOARD_BUILD_TESTS "Build offline database/security tests" ON)
if(QUICKCLIPBOARD_BUILD_TESTS)
  enable_testing()
  add_executable(quickclipboard_tests tests/core_tests.cpp)
  add_dependencies(quickclipboard_tests QuickClipboard)
  target_link_libraries(quickclipboard_tests PRIVATE quickclipboard_core psapi)
  add_test(NAME core COMMAND quickclipboard_tests)
endif()
install(TARGETS QuickClipboard RUNTIME DESTINATION .)
